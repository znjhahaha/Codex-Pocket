cmake_minimum_required(VERSION 3.22)
project(pocket_p2p_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# libdatachannel's MbedTLS DTLS backend references this API even with NO_MEDIA.
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
if(NOT DEFINED POCKET_P2P_DEPS)
  message(FATAL_ERROR "POCKET_P2P_DEPS must point to verified pinned dependency sources")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${POCKET_P2P_DEPS}/mbedtls" mbedtls EXCLUDE_FROM_ALL)
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(USE_NICE OFF CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${POCKET_P2P_DEPS}/libdatachannel" libdatachannel EXCLUDE_FROM_ALL)
add_library(pocket-p2p SHARED src/transport.cpp)
target_compile_definitions(pocket-p2p PRIVATE POCKET_P2P_BUILD)
target_include_directories(pocket-p2p PUBLIC include)
target_link_libraries(pocket-p2p PRIVATE datachannel)
if(WIN32)
  add_custom_command(TARGET pocket-p2p POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:datachannel> $<TARGET_FILE_DIR:pocket-p2p>)
endif()
if(ANDROID OR POCKET_P2P_JNI)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
  endif()
  add_library(pocket-p2p-jni SHARED
    "${CMAKE_CURRENT_SOURCE_DIR}/../../android/wire/src/main/cpp/DataChannelJni.cpp")
  target_include_directories(pocket-p2p-jni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(pocket-p2p-jni PRIVATE pocket-p2p)
endif()
if(NOT ANDROID)
  add_executable(pocket-p2p-stream-fixture stream_fixture.cpp)
  target_link_libraries(pocket-p2p-stream-fixture PRIVATE pocket-p2p)
  enable_testing()
  add_test(NAME stream-fixture COMMAND pocket-p2p-stream-fixture)
  set_tests_properties(stream-fixture PROPERTIES TIMEOUT 60)
  add_executable(pocket-p2p-probe probe.cpp)
  target_link_libraries(pocket-p2p-probe PRIVATE datachannel)
  add_custom_command(TARGET pocket-p2p-probe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:datachannel> $<TARGET_FILE_DIR:pocket-p2p-probe>)
endif()
